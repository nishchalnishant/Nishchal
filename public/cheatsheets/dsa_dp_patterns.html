    Complexity analysis toolkit#

    T(n) = a·T(n/b) + f(n) compare f(n) with n^(log_b a)f smaller → Θ(n^(log_b a))f equal → Θ(n^(log_b a) · log n)f larger → Θ(f(n))2T(n/2) + O(1) → Θ(n) tree traversal2T(n/2) + O(n) → Θ(n log n) merge sortT(n/2) + O(1) → Θ(log n) binary searchT(n−1) + T(n−2) → Θ(φⁿ) naive Fibonacci
    • Amortised: dynamic array append O(1) (doubling), two-stack queue O(1), union-find ≈ O(α(n)), monotonic stack O(n) total because each item is pushed and popped once.
    • Recursion space = max depth × frame size; memo tables add the state count.
    • Backtracking: branching^depth × work per node (subsets 2ⁿ·n, permutations n!·n).
    • Graph: adjacency list O(V+E) space; BFS/DFS O(V+E); Dijkstra O((V+E) log V).
    • Strings: slicing and concatenation in a loop are O(n) each → O(n²) total; use a list and join.
    • Hash maps: O(1) average, O(n) worst case; sorting is the safe O(n log n) alternative.

    Sliding window (variable size)#

    Longest or shortest contiguous range that satisfies a constraint that can be repaired by shrinking.

    def longest_at_most_k_distinct(s, k):
    count, best, left = {}, 0, 0
    for right, ch in enumerate(s):
        count[ch] = count.get(ch, 0) + 1
        while len(count) > k:              # shrink until valid again
            c = s[left]
            count[c] -= 1
            if count[c] == 0: del count[c]
            left += 1
        best = max(best, right - left + 1)
    return best
    static int longestAtMostKDistinct(String s, int k) {
    Map<Character, Integer> count = new HashMap<>();
    int best = 0, left = 0;
    for (int right = 0; right < s.length(); right++) {
        count.merge(s.charAt(right), 1, Integer::sum);
        while (count.size() > k) {
            char c = s.charAt(left++);
            if (count.merge(c, -1, Integer::sum) == 0) count.remove(c);
        }
        best = Math.max(best, right - left + 1);
    }
    return best;
}

    BFS: shortest path on a grid#

    Unweighted shortest path, minimum steps, multi-source spread (seed the queue with every source).

    from collections import deque

def shortest_path(grid, start, goal):   # grid[r][c] == 1 is a wall
    R, C = len(grid), len(grid[0])
    dist = {start: 0}
    q = deque([start])
    while q:
        r, c = q.popleft()
        if (r, c) == goal: return dist[(r, c)]
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < R and 0 <= nc < C and grid[nr][nc] == 0 and (nr, nc) not in dist:
                dist[(nr, nc)] = dist[(r, c)] + 1   # mark when enqueued
                q.append((nr, nc))
    return -1
    static int shortestPath(int[][] g, int sr, int sc, int tr, int tc) {
    int R = g.length, C = g[0].length;
    int[][] dist = new int[R][C];
    for (int[] row : dist) Arrays.fill(row, -1);
    Deque<int[]> q = new ArrayDeque<>();
    dist[sr][sc] = 0; q.offer(new int[]{sr, sc});
    int[][] dirs = {{1,0},{-1,0},{0,1},{0,-1}};
    while (!q.isEmpty()) {
        int[] cur = q.poll();
        if (cur[0] == tr && cur[1] == tc) return dist[tr][tc];
        for (int[] d : dirs) {
            int nr = cur[0] + d[0], nc = cur[1] + d[1];

    Topological sort (Kahn's algorithm)#

    Prerequisites, build order, course schedule; leftover nodes mean a cycle.

    from collections import deque

def topo_order(n, edges):           # (u, v): u must come before v
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for u, v in edges:
        adj[u].append(v); indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    order = []
    while q:
        u = q.popleft(); order.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0: q.append(v)
    return order if len(order) == n else []   # [] means a cycle
    static List<Integer> topoOrder(int n, int[][] edges) {
    List<List<Integer>> adj = new ArrayList<>();
    for (int i = 0; i < n; i++) adj.add(new ArrayList<>());
    int[] indeg = new int[n];
    for (int[] e : edges) { adj.get(e[0]).add(e[1]); indeg[e[1]]++; }
    Deque<Integer> q = new ArrayDeque<>();
    for (int i = 0; i < n; i++) if (indeg[i] == 0) q.offer(i);
    List<Integer> order = new ArrayList<>();
    while (!q.isEmpty()) {
        int u = q.poll(); order.add(u);
        for (int v : adj.get(u)) if (--indeg[v] == 0) q.offer(v);
    }
    return order.size() == n ? order : List.of();      // empty means a cycle

    Dijkstra's shortest paths#

    Single-source shortest paths with non-negative weights.

    import heapq

def dijkstra(n, adj, src):          # adj[u] = [(v, w), ...], w >= 0
    dist = [float('inf')] * n
    dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]: continue    # stale entry
        for v, w in adj[u]:
            if d + w < dist[v]:
                dist[v] = d + w
                heapq.heappush(pq, (dist[v], v))
    return dist
    static long[] dijkstra(int n, List<List<int[]>> adj, int src) {  // int[]{v, w}
    long[] dist = new long[n];
    Arrays.fill(dist, Long.MAX_VALUE);
    dist[src] = 0;
    PriorityQueue<long[]> pq = new PriorityQueue<>(Comparator.comparingLong(x -> x[0]));
    pq.offer(new long[]{0, src});
    while (!pq.isEmpty()) {
        long[] top = pq.poll();
        int u = (int) top[1];
        if (top[0] > dist[u]) continue;                 // stale entry
        for (int[] e : adj.get(u)) {
            long nd = top[0] + e[1];
            if (nd < dist[e[0]]) { dist[e[0]] = nd; pq.offer(new long[]{nd, e[0]}); }
        }
    }
    return dist;

    Union-Find (disjoint set union)#

    Connected components, cycle detection in undirected graphs, Kruskal's MST, grouping accounts.

    class DSU:
    def __init__(self, n):
        self.p = list(range(n))
        self.sz = [1] * n
    def find(self, x):
        while self.p[x] != x:
            self.p[x] = self.p[self.p[x]]   # path halving
            x = self.p[x]
        return x
    def union(self, a, b):
        a, b = self.find(a), self.find(b)
        if a == b: return False             # already connected
        if self.sz[a] < self.sz[b]: a, b = b, a
        self.p[b] = a
        self.sz[a] += self.sz[b]
        return True
    final class DSU {
    private final int[] p, sz;
    DSU(int n) { p = new int[n]; sz = new int[n]; for (int i = 0; i < n; i++) { p[i] = i; sz[i] = 1; } }
    int find(int x) {
        while (p[x] != x) { p[x] = p[p[x]]; x = p[x]; }   // path halving
        return x;
    }
    boolean union(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) { int t = a; a = b; b = t; }
        p[b] = a; sz[a] += sz[b];
        return true;
    }
}

    Backtracking: combination sum#

    Enumerate subsets, combinations, permutations; choose, recurse, undo, and prune early.

    def combination_sum(cands, target):  # each candidate can be reused
    cands.sort()
    res, path = [], []
    def dfs(start, remain):
        if remain == 0:
            res.append(path[:]); return
        for i in range(start, len(cands)):
            if cands[i] > remain: break   # sorted, so prune the rest
            path.append(cands[i])
            dfs(i, remain - cands[i])     # i, not i + 1: reuse allowed
            path.pop()                    # undo

    return res

static void dfs(int[] c, int start, int remain, List<Integer> path, List<List<Integer>> res) {
    if (remain == 0) { res.add(new ArrayList<>(path)); return; }
    for (int i = start; i < c.length && c[i] <= remain; i++) {
        path.add(c[i]);
        dfs(c, i, remain - c[i], path, res);       // i: reuse allowed
        path.remove(path.size() - 1);              // undo
    }

    DSU + Kruskal ≈O(α(n)) per op; Kruskal O(E log E)#

    parent = list(range(n)); rank = [0]*n   # NOT [0]*n
def find(x):
    while parent[x] != x:
        parent[x] = parent[parent[x]]; x = parent[x]
    return x
def union(a, b):
    a, b = find(a), find(b)
    if a == b: return False          # already connected
    if rank[a] < rank[b]: a, b = b, a
    parent[b] = a
    if rank[a] == rank[b]: rank[a] += 1
    return True

    Common mistakes (practice/graph.md, dsu.md)#

    • Forgetting disconnected graphs — always outer-loop over all nodes
    • Marking visited on dequeue instead of enqueue → duplicates, wrong distances
    • Parent-tracking DFS on a directed graph — use 3-state DFS instead
    • Dijkstra without if d > dist[u]: continue
    • Topo sort without the len(order) == n check
    • Bellman-Ford updating dist in place — copy before each round
    • DSU: parent = [0]*n; forgetting return False when a == b
    • DFS for shortest paths in an unweighted graph: use BFS
    • Dijkstra with negative edges: use Bellman-Ford
    • Recursive DFS on 105 nodes in Python hits the recursion limit: go iterative
    • Undirected graph built with only one direction of each edge
    • Union-find without path compression and union by rank degrades to O(n) per operation

    Fenwick tree (practice/data_structures/11)#

    class FenwickTree:            # 1-indexed inside
    def __init__(self, n): self.n = n; self.tree = [0]*(n+1)
    def update(self, i, delta):
        i += 1
        while i <= self.n: self.tree[i] += delta; i += i & -i
    def prefix_sum(self, i):    # sum [0, i]
        i += 1; s = 0
        while i > 0: s += self.tree[i]; i -= i & -i
        return s
